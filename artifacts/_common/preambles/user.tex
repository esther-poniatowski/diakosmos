%% artifacts/_common/preambles/user.tex
%% Common preamble for all documents in this workspace.
%%
%% Loads external packages, then the scholia educational formatting package.
%% Scholia auto-detects the document class (article or beamer).
%%
%% PREREQUISITE: \RootPath must be defined in the calling document,
%% pointing to the workspace root relative to the document's location.
%%
%% USAGE:
%%   \documentclass{article}          % or \documentclass{beamer}
%%   \newcommand{\RootPath}{../../..}  % adjust depth as needed
%%   \input{\RootPath/artifacts/_common/preambles/user}
%%   \begin{document}...
%%
%% ==================================================================================================

\makeatletter

% --- External Dependencies ---
\input{\RootPath/artifacts/_common/preambles/packages}

% --- Scholia Package ---
% Educational document formatting (callouts, exercises, etc.)
% Beamer class is auto-detected by scholia.
\usepackage[english]{\RootPath/vendor/scholia/scholia}

% --- Icon Shortcuts ---
% Concise commands for frequently used inline icons (built on \scholiaicon).
\newcommand{\downarrowblack}{\scholiaicon{minimalist-down-arrow-black.pdf}}
\newcommand{\downarrowwhite}{\scholiaicon{minimalist-down-arrow-white.pdf}}
\newcommand{\twoheadsearrow}{\scholiaicon{minimalist-bi-arrow-horizontal.pdf}}
\newcommand{\droparrow}{\scholiaicon{minimalist-right-curve-arrow.pdf}}
\newcommand{\lightbulb}{\scholiaicon[height=1.8em]{minimalist-light-bulb.pdf}}
\newcommand{\observation}{\scholiaicon{minimalist-observation.pdf}}
\newcommand{\paper}{\scholiaicon[height=0.9em]{minimalist-paper-1.pdf}}

% --- Default Figure Path ---
% Documents conventionally place figures in a figures/ subdirectory.
\graphicspath{{figures/}}

\makeatother
