% Typography of the house style (LuaLaTeX + fontspec). The faces are resolved by name from the TeX
% Live tree (sourcesans, sourcecodepro and raleway packages). Two looks share one roman, one code face
% and one math text face:
%   - DEFAULT  the final look of the posters (the Royaumont A0 sheets, the CCN supplement, the landing
%              pages): Source Sans 3 throughout, with Semibold as its bold; the title and heading
%              faces are the sans family.
%   - RALEWAY  the look of the printed CCN 2026 sheet: Raleway for the title (Medium) and the headings
%              (SemiBold and Bold), Source Sans 3 with Bold as its bold. A document selects it by
%              defining \FontVariant as `raleway' before it inputs this file.
% The MATH SIZES belong to each document's type scale (artifacts/_common/style/type-roles.tex).

\usepackage{fontspec}
\defaultfontfeatures{Ligatures=TeX}

% --- Body / default family: Source Sans 3 (sans) ---
% The ROMAN font is Latin Modern (serif) so that math alphabets (\mathrm, \mathbf) and the
% math "operators" font are serif, consistent with the serif math symbols. The document default
% is still sans (familydefault = sfdefault), so body text is Source Sans 3.
\setmainfont{Latin Modern Roman}
\renewcommand{\familydefault}{\sfdefault}

\providecommand\FontVariant{default}
\def\FontVariantRaleway{raleway}
\ifx\FontVariant\FontVariantRaleway
  \setsansfont{SourceSans3-Regular.otf}[BoldFont=SourceSans3-Bold.otf,
    ItalicFont=SourceSans3-RegularIt.otf, BoldItalicFont=SourceSans3-BoldIt.otf]
  % --- Display family: Raleway (weights addressed explicitly) ---
  % Title uses a lighter Medium weight (Black reads too heavy at poster scale).
  \newfontfamily\titlefont{Raleway Medium}
  \newfontfamily\headingfont{Raleway SemiBold}[
    BoldFont       = {Raleway Bold},
    ItalicFont     = {Raleway SemiBold Italic},
    BoldItalicFont = {Raleway Bold Italic}
  ]
\else
  \setsansfont{SourceSans3-Regular.otf}[BoldFont=SourceSans3-Semibold.otf,
    ItalicFont=SourceSans3-RegularIt.otf, BoldItalicFont=SourceSans3-SemiboldIt.otf]
  \newcommand{\titlefont}{\sffamily}
  \newcommand{\headingfont}{\sffamily}
\fi

% --- Supporting body weights ---
\newfontfamily\bodysemibold{SourceSans3-Semibold.otf}

% --- Mono family for section-heading kickers / metadata, and its Semibold for the role names (the
% kicker face of artifacts/mixture-input-encoding_poster_ccn2026/inspiration/claude-design-layout.tex) ---
\newfontfamily\kickerfont{Source Code Pro}
\newfontfamily\kickerbold{SourceCodePro-Semibold.otf}

% --- Math text font: the body sans, so descriptive WORDS inside a formula (reweighting, projection,
% comp. plane, ...) match the body prose. Symbols, vectors, matrices and the "mix" indices keep the
% math font. Used via \mlab (artifacts/_common/style/type-roles.tex). ---
\newfontfamily\mathtextfont{SourceSans3-Regular.otf}
