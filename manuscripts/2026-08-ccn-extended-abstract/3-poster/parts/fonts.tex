% Typography (LuaLaTeX + fontspec). Two families realise the plan's hierarchy:
%   - Raleway          : display geometric sans -> title (Black) and headings (SemiBold/Bold)
%   - Source Sans Pro  : neutral sans          -> body, captions, labels, tables
% Fonts are resolved by name from the TeX Live tree (raleway, sourcesanspro packages).

\usepackage{fontspec}
\defaultfontfeatures{Ligatures=TeX}

% --- Body / default family: Source Sans Pro (sans) ---
% The ROMAN font is Latin Modern (serif) so that math alphabets (\mathrm, \mathbf) and the
% math "operators" font are serif, consistent with the serif math symbols. The document default
% is still sans (familydefault = sfdefault), so body text is Source Sans Pro.
\setmainfont{Latin Modern Roman}
\setsansfont{Source Sans Pro}
\renewcommand{\familydefault}{\sfdefault}

% --- Display family: Raleway (weights addressed explicitly) ---
% Title uses a lighter Medium weight (Black reads too heavy at poster scale).
\newfontfamily\titlefont{Raleway Medium}
\newfontfamily\headingfont{Raleway SemiBold}[
  BoldFont       = {Raleway Bold},
  ItalicFont     = {Raleway SemiBold Italic},
  BoldItalicFont = {Raleway Bold Italic}
]

% --- Supporting body weights ---
\newfontfamily\bodysemibold{Source Sans Pro Semibold}

% --- Mono family for section-heading kickers / metadata (per tests/claude-design-layout.tex) ---
\newfontfamily\kickerfont{Source Code Pro}

% --- Math text font: the body sans (Source Sans Pro), so descriptive words inside formulas
% (amplification, projection, in span, ...) match the body prose. Math symbols, the m/n vectors,
% matrices, and the "mix" indices keep their math font. Used via \mlab (see macros.tex). ---
\newfontfamily\mathtextfont{Source Sans Pro}

% --- MATH SIZES, ONE DECLARATION PER STEP OF THE TYPE SCALE ---------------------------------------
% \DeclareMathSizes{<text size>}{<math>}{<script>}{<scriptscript>}. It fires when the TEXT size is set
% -- including inside a TikZ node, once the size is one of these -- so there must be a line here for
% EVERY step of the scale in macros.tex that can carry math. A step with no line falls back to LaTeX's
% defaults and drifts out of the system silently.
%
% MATH = 1.125 x TEXT. The body is Source Sans Pro; math is Latin Modern (serif), whose smaller
% x-height makes a symbol look smaller than the sans text beside it at the same nominal size. The
% 12.5% bump matches the x-heights, so an inline state or matrix reads level with the prose.
%
% SCRIPT / SCRIPTSCRIPT are the conventional 0.7 / 0.5 of the math size (a deliberate choice: the
% floors in macros.tex govern TEXT, not indices). Consequence to know about: at body size the "mix"
% index of u_mix / kappa_mix lands at 23 pt -- under the 24 pt figure floor, and the smallest mark on
% the poster. If it proves unreadable at viewing distance, raise the last two numbers here; nothing
% else needs to change.
\DeclareMathSizes{24}{27}{19}{14}      % \figticksize  -- figure floor
\DeclareMathSizes{30}{34}{24}{17}      % \figannotsize
\DeclareMathSizes{36}{40}{28}{20}      % \notesize / \figcapsize -- text floor
\DeclareMathSizes{40}{45}{32}{23}      % \bodysize / \bodycompact -- incl. the two move (a) formulae
\DeclareMathSizes{44}{50}{35}{25}      % \cardtitlesize
\DeclareMathSizes{52}{58}{41}{29}      % \headingsize
% The brace readings under the move (a) formulae do NOT follow these sizes -- they are sans, pinned by
% \brlab (macros.tex), because a gloss is words, not algebra.
