% THE COLOUR SYSTEM'S ENTRY POINT, and the only file either document inputs (parts/preamble.tex and
% supplement/preamble.tex both say `\input{parts/colors}'). It selects a scheme and loads the three
% layers in the one order that works.
%
% >>> THREE LAYERS, AND THE ORDER IS LOAD-BEARING <<<
%   1. VALUES  parts/colors-values.tex  -- what each token IS. Every \definecolor.
%   2. SCHEME  schemes/<name>.tex       -- the variant's overrides: values, and the switches below.
%   3. ROLES   parts/colors-roles.tex   -- where each colour GOES. Every \colorlet.
% \colorlet COPIES A VALUE at the moment it is read. Run the roles before the scheme and every alias
% keeps its base value: half the page repaints, half does not, and NOTHING IN THE LOG SAYS SO. That
% failure is the entire reason these are three files instead of one.
%
% >>> SELECTING A SCHEME <<< \posterscheme names a file in schemes/. It is \providecommand'd, so the
% default survives an ordinary build while a \def injected before the document wins. Ktisma owns that
% injection: each variant in .ktisma.toml carries the \def as its payload.
%     ktisma variants --include-default mixture-geometry-poster.tex   # base + every variant
%     ktisma build --variant neutral-chrome mixture-geometry-poster.tex
% The default lives HERE rather than in either driver, so the supplement inherits the same scheme for
% free and cannot drift out of step with the poster it belongs to.
\providecommand\posterscheme{base}

\input{parts/colors-values}

% ----- SCHEME SWITCHES: declared before the scheme, so the scheme may set them --------------------
% A switch is for a variation that is NOT expressible as a colour value -- a change of TREATMENT.
% Keep this list short: a switch is a fork in the presentation code, and every fork is a pair of
% renders somebody has to check. Everything a colour can express belongs in a value or a role.
%   \ifquestioninverted -- do the two question boxes wear an accent GROUND (true, the base) or the
%                          ordinary card ground with an accent ring (false)? Everything the choice
%                          implies is derived in parts/colors-roles.tex and parts/block-columns.tex.
\newif\ifquestioninverted\questioninvertedtrue

\input{schemes/\posterscheme}

% THE FIFTH MEASUREMENT CHANNEL. The other four report geometry; this one reports WHICH POSTER THIS
% IS. Without it a dropped -pretex is invisible: every variant builds the base, and a set of
% identical PDFs is indistinguishable from a set of successful variants until somebody compares two
% of them by eye. scripts/check-build.sh compares this line against the build directory ktisma named,
% which is where the REQUESTED scheme is recorded.
\typeout{[scheme] \posterscheme}

\input{parts/colors-roles}
