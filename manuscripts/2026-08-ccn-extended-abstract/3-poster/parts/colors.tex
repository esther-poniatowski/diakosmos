% Poster palette, three INDEPENDENT palettes (cf. poster-plan.md > Style > Colors). Each colour token
% belongs to exactly ONE palette, so any palette can be retuned without shifting the other two:
%   1. Neutral foundation — the achromatic substrate: page background, surfaces, borders, text levels,
%                           and the neutral structure of plots (axes, reference rules). Shared by
%                           chrome AND figures, but it is NEVER an accent and NEVER encodes a concept,
%                           and no design/semantic hue is derived from it.
%   2. Design chrome      — turquoise, plus a RESERVED complementary coral (accentcoral): headings,
%                           boxes, navigation, QR tags, badges, the Approach timeline, AND the two
%                           kinds of POINTER AWAY from the poster (literature citations \litcite;
%                           supporting-detail tags \seebottom) -- one colour, because they are one
%                           kind of mark. NEVER used to encode a concept inside a figure.
%   3. Semantic grammar   — one concept -> one colour, used ONLY inside figures/representations. The
%                           semantic coral (dncoral) is a SEPARATE token from the design coral above
%                           -- same value, independently tunable. NEVER for chrome.
% Values are sRGB approximations of the OKLCH specification; the perceptual ladder
% (relative lightness ordering) is what matters, not exact colorimetry.

% ----- 1. Neutral foundation (achromatic substrate; independent of both accent palettes) -----
\definecolor{posterbg}{HTML}{F1F3F3}      % L=0.970  background
\definecolor{surface}{HTML}{FBFCFC}       % L=0.995  elevated panels (shadow separates from bg)
\definecolor{bordersubtle}{HTML}{DBDFDF}  % L=0.900  decorative outlines, dividers, table lines
\definecolor{borderfunc}{HTML}{8C9698}    % L<=0.640 focus rings, emphasis boundaries
\definecolor{texttertiary}{HTML}{6E797B}  % L=0.530  captions, minor metadata, faint hints
\definecolor{textsecondary}{HTML}{586366} % L=0.450  subtitles, supporting text
\definecolor{textprimary}{HTML}{1B2224}   % L=0.150  titles, labels, body, key values

% ----- 2. Design-chrome accent (headings, boxes, navigation) -- NEVER used inside a figure -----
% Primary = turquoise (all section headings/boxes, QR tags, citations, the Approach timeline, badges).
% A complementary coral is RESERVED for design emphasis (currently unused). It is a DESIGN token,
% deliberately distinct from the semantic coral (dncoral) even though it shares the value today --
% so chrome and figures remain independently tunable.
\definecolor{accentturq}{HTML}{12A89E}    % primary design accent, H_acc ~ 185; also \litcite + \seebottom
\definecolor{accentturqdk}{HTML}{0C7A72}   % darker turquoise: sub-headers, contact line, badges
\definecolor{accentcoral}{HTML}{F26B5B}   % complementary DESIGN accent -- RESERVED (chrome only; != dncoral)

% ----- 3. Semantic / concept grammar (one concept -> one colour) -- used ONLY inside figures -----
\definecolor{netturq}{HTML}{2BB3A6}       % concrete network with neurons
\definecolor{inputpurple}{HTML}{8358C6}   % external input  AND  linear-projection stage
\definecolor{recurjade}{HTML}{1F8E7E}     % recurrent circuit / amplification stage / circuit prediction
% --- Approach-figure palette, matching the low-rank-model-introduction deck -----------------------
\definecolor{inputorange}{RGB}{255,127,14}    % input  (Rate model)
\definecolor{orangeA}{RGB}{230,90,20}         % input pattern 1  I^(1)
\definecolor{orangeB}{RGB}{220,170,40}        % input pattern 2  I^(2)
\definecolor{recurrentgreen}{RGB}{44,160,44}  % connectivity matrix J  (Low-rank connectivity)
\definecolor{greenA}{RGB}{0,128,90}           % recurrent pattern 1 / recurrence  (Low-rank connectivity)
\definecolor{greenB}{RGB}{120,185,50}         % recurrent pattern 2  m_2 n_2
\definecolor{collectivepurple}{RGB}{148,103,189} % response kappa / state x
\definecolor{recpoolcyan}{RGB}{23,190,207}    % recurrent pooling Sigma^(mm)  (cold)
\definecolor{inpoolwarm}{RGB}{214,118,52}     % input pooling Sigma^(II)  (warm)
\definecolor{gainblue}{HTML}{2E6FC4}      % gain g(Delta), total-input variance
\definecolor{compred}{HTML}{D6453C}       % component response 1 (kappa_1)
\definecolor{compgreen}{HTML}{4F9A3F}     % component response 2 (kappa_2)
\definecolor{mixamber}{HTML}{F2A30C}      % collective / mixture response (kappa_mix)
\definecolor{spangray}{HTML}{B7BFC1}      % component span plane / proportional reference
\definecolor{emergentgray}{HTML}{566063}  % emergent-axis annotations (bias arc, in-plane foot marker)
\definecolor{emergentbrown}{HTML}{8A5524} % orthogonal component (eta_perp) -- solid brown, all figures
\definecolor{dncoral}{HTML}{F26B5B}       % divisive-normalization rival model (DN card + competition curve)

\colorlet{projstage}{inputorange}         % projection stage = input / feedforward (orange)
\colorlet{ampstage}{greenA}               % amplification stage = recurrence (green)
