% Reusable toolkit: type sizes, math macros, layout grid, coordinate helpers,
% card / heading / icon / citation commands, and TikZ arrow styles.
% Requires colors.tex and fonts.tex to be loaded first.

% ===== THE TYPE SYSTEM ============================================================================
% The poster is 147.32 x 172.72 cm at TRUE SCALE: 1 pt = 0.351 mm, so a point size IS a physical
% size. 40 pt = a 14.1 mm em; 24 pt = 8.4 mm. Read every number below as a millimetre statement.
%
% THIS FILE IS THE ONLY PLACE A POINT SIZE MAY BE CHOSEN. Do not write \fontsize{..}{..} anywhere
% else -- not in a block, not in a figure, not inside a TikZ node. The previous system declared a
% hierarchy that nothing used (seven of its fifteen macros were never called) while ~90 call sites
% hard-coded 29 different sizes and 20 different leadings between them. If a role needs a size that
% is not here, add a NAMED macro here; do not inline it there.
%
% ----- Two readability floors -------------------------------------------------------------------
% TEXT (anything a visitor reads as language: prose, headings, captions, tags, notes, references,
%   card content -- INCLUDING text drawn inside a TikZ card):        floor = 36 pt  (12.7 mm em)
% FIGURE (annotations inside a diagram or plot: axis ticks, vector
%   names, coefficient marks, symbol labels):                        floor = 24 pt  ( 8.4 mm em)
% The two floors differ because they are read at different distances: prose is read across the aisle,
% a figure annotation is read while standing at the poster. Nothing may be set below its own floor.
%
% ----- Font size is ABSOLUTE inside figures, and that is deliberate ------------------------------
% Figures scale their COORDINATES (`scale=1.4') and never their fonts: no `transform shape', no
% \scalebox around a picture. A figure may therefore be resized freely -- to fit a column, to fill a
% gap -- WITHOUT its labels shrinking below the floor. Keep it that way: the moment fonts scale with
% the drawing, legibility becomes a function of layout accident.
%
% ----- Three leadings, not one ------------------------------------------------------------------
%   PROSE    1.40x -- running text that wraps for several lines. Aerated, for reading at distance.
%   COMPACT  1.15x -- headings, captions, labels, notes: text that wraps to two or three lines and
%                     must stay a visual unit. A 1.4 leading pulls such a block apart.
%   FLAT     1.00x -- single-line nodes and icon glyphs, where leading is inert. Set = size so the
%                     value is predictable rather than accidental.
%
% ----- The scale: 8 point sizes ------------------------------------------------------------------
% TEXT (floor 36)
\newcommand{\titlesize}{\fontsize{94}{108}\selectfont}      % poster title

\newcommand{\headingsize}{\fontsize{52}{60}\selectfont}     % block headings, authors, display formulae

\newcommand{\bodysize}{\fontsize{40}{56}\selectfont}        % prose that wraps            (PROSE 1.40)
\newcommand{\bodycompact}{\fontsize{40}{46}\selectfont}     % body size, tight            (COMPACT)

% outside one are the same kind of small text, and one value keeps them from drifting apart.

% In-math text label rendered in the math (serif) text font, so words match the symbols.
\newcommand{\mlab}[1]{\text{{\mathtextfont #1}}}
% An \underbracket's READING, pinned to 32 pt INDEPENDENTLY of the formula's size. \mlab would follow
% the surrounding math size, so enlarging a formula for emphasis would enlarge its gloss with it -- and
% the gloss is not part of the statement: it is the words that read the statement, and it must stay
% quiet. Use \brlab (not \mlab) under a brace whenever the formula is set above body size.
\newcommand{\brlab}[1]{\text{{\mathtextfont\notesize #1}}}

% ===== Card (panel) styles ========================================================================
% UX-inspired cards: subtle surface contrast + soft shadow, no loud frame.
\tcbset{
  % shadow-free base: surface, rounded corners, padding.
  % CORNER RADIUS IS A RATIO, NOT A NUMBER. 24 mm on a 65 cm card is ~3.7% of its width; the reference
  % UI cards this style follows sit at 4-6%. The old 15 mm was 2.3%, and at poster scale a 2% arc is a
  % chamfer, not a radius -- the cards read square. The nested cards (mechanism, model comparison) carry
  % 14 mm on their ~30 cm width, i.e. the same ~4.6%: SCALE THE RADIUS WITH THE CARD, do not copy the
  % number across. See `stagecard' in figures/results-matrix.tex and \modelcard in block-consequences.
  cardbase/.style={
    enhanced, arc=24mm, outer arc=24mm,
    colback=surface, colframe=bordersubtle, boxrule=0.8pt,
    boxsep=0pt, left=12mm, right=12mm, top=11mm, bottom=11mm,
    fontupper=\color{textprimary}\bodysize,
  },
  % ===== ONE SHADOW FOR EVERY CARD ON THE POSTER ==================================================
  % EVERY card now GLOWS in its own colour, and every card uses the SAME BLUR -- the same primitive,
  % the same cast, the same spread. Only the TINT changes.
  %
  % THE PRIMITIVE MATTERS AS MUCH AS THE NUMBERS, and this is the trap that cost a round: tcolorbox's
  % `fuzzy shadow' is a STACK OF OFFSET ROUNDED RECTANGLES, whereas TikZ's `blur shadow' (shadows.blur)
  % is a TRUE GAUSSIAN BLUR. Giving a fuzzy shadow the blur shadow's NUMBERS changes nothing -- the box
  % still renders with the stepped halo. So the blur is attached to the tcolorbox's FRAME PATH, which is
  % the outermost path and is drawn beneath the interior: the shadow lands behind the whole card.
  % The numbers are \mechanismcards' verbatim (figures/results-matrix.tex > projglow/ampglow):
  % yshift -2.6mm, radius 6mm, 12 steps, opacity 40. CHANGE THEM IN ONE PLACE, HERE, OR NOWHERE.
  %
  % THE TINT GOES THROUGH `every shadow'. A `fill=' inside `blur shadow={...}' LEAKS onto the path it is
  % attached to -- here the frame -- and the card comes out flooded with the tint colour.
  %   #1 = the ring colour (the frame path must keep filling itself with it)
  %   #2 = the tint of the glow
  cardglow/.style 2 args={
    frame style={fill=#1,
      blur shadow={shadow xshift=0pt, shadow yshift=-2.6mm, shadow blur radius=6mm,
        shadow blur steps=12, shadow opacity=40},
      every shadow/.style={fill=#2, draw=none}}},
  % THE SECTION CARDS GLOW TURQUOISE, NOT GREY -- and NOT in their own ring, which is `bordersubtle', a
  % near-white hairline whose glow would be invisible. Their IDENTITY is the chrome turquoise (heading,
  % number disc, icon), so that is what they cast. To go back to a neutral shadow: `cardglow={bordersubtle}{cardshadow}'.
  card/.style={cardbase, cardglow={bordersubtle}{accentturq}},
  % Emphasis card (Key results, Open Question). Emphasis is RELIEF, not tint: a pronounced accent ring
  % (3 pt, the heaviest on the poster) over a white interior, so the text on it reads exactly like the
  % text on every other card. It glows in the same turquoise as the section cards -- what separates it
  % is the RING WEIGHT, not the shadow, now that every card glows.
  % TO RESTORE THE OLD GRADIENT FILL: the two call sites keep it commented out -- \yQuest's box in
  % block-starting.tex (`interior style`) and the title band in block-takeaway.tex (`underlay`).
  emphcard/.style={cardbase, colback=surface, colframe=accentturq, boxrule=3pt,
    cardglow={accentturq}{accentturq}},
  % Kept as a NAME so the Open Question's call site still reads as "the box that must not be missed",
  % but it is now the same card: every card glows, so the glow no longer distinguishes it. Its ring does.
  emphcardglow/.style={emphcard},
  plaincard/.style={
    enhanced, arc=24mm, outer arc=24mm,
    colback=posterbg, colframe=posterbg, boxrule=0pt,
    boxsep=0pt, left=10mm, right=10mm, top=8mm, bottom=8mm,
    fontupper=\color{textprimary}\bodysize,
  },
  % transparent panel: same padding/alignment as a card, but no fill, frame or shadow
  % (content lies directly over the page background)
  ghostcard/.style={
    enhanced, frame hidden, interior hidden,
    boxsep=0pt, left=12mm, right=12mm, top=11mm, bottom=11mm,
    fontupper=\color{textprimary}\bodysize,
  },
  % dashed outline box (no fill, no shadow): frames the two self-consistent-gain items
  dashbox/.style={enhanced, interior hidden, frame hidden,
    arc=5mm, outer arc=5mm, boxsep=0pt, left=4mm, right=4mm, top=5mm, bottom=5mm,
    borderline={1.1pt}{0pt}{texttertiary!55, dashed}},
}

% A tcolorbox typeset directly inside a TikZ \node triggers an infinite NFSS
% font-selection loop under tikzposter + fontspec/LuaLaTeX. The fix is two-step:
% build the card into a savebox at document top level (OUTSIDE any tikzpicture),
% then place that box with \node{\usebox} inside the overlay picture.
%
% Build card CONTENT into <savebox> (call at document top level):
%   \buildcard{<\savebox>}{<width cm>}{<height cm>}{<tcb style>}{<content>}
\newcommand{\buildcard}[5]{%
  \sbox#1{\begin{tcolorbox}[#4, width=#2 cm, height=#3 cm]#5\end{tcolorbox}}%
}
% Build arbitrary (card-free) content into <savebox> at document top level:
%   \buildbox{<\savebox>}{<content>}
\newcommand{\buildbox}[2]{\sbox#1{#2}}
% Build a fixed-width text column (minipage) into <savebox>:
%   \buildcol{<\savebox>}{<width cm>}{<content>}
\newcommand{\buildcol}[3]{\sbox#1{\begin{minipage}[t]{#2 cm}#3\end{minipage}}}

% Drop a pre-built box with its NORTH-WEST corner at (x,y) cm (call INSIDE the picture):
%   \dropbox{<\savebox>}{<x>}{<y>}
\newcommand{\dropbox}[3]{%
  \node[anchor=north west, inner sep=0pt] at \xy{#2}{#3} {\usebox#1};%
}
% As \dropbox but with an explicit anchor (default north west):
%   \dropboxa[<anchor>]{<\savebox>}{<x>}{<y>}
\newcommand{\dropboxa}[4][north west]{%
  \node[anchor=#1, inner sep=0pt] at \xy{#3}{#4} {\usebox#2};%
}

% ===== Icon badge + block heading =================================================================
% Circular accent badge holding a FontAwesome glyph (or any content).
%   #1 fill color  #2 glyph color  #3 glyph
\newcommand{\iconbadge}[3]{%
  \tikz[baseline=-0.42ex]{%
    \node[circle, fill=#1, text=#2, minimum size=20mm, inner sep=0pt,
          font=\iconmd]{#3};}%
}

% Functional block heading: a soft rounded-square icon (light accent tint, dark accent glyph) wearing
% its SECTION NUMBER on a small solid disc at the top-right corner, + a MONO accent kicker + a
% title-font descriptive sentence.
%   [#1 sentence font, optional]  #2 accent  #3 glyph  #4 SECTION NUMBER  #5 KICKER  #6 sentence
% The number is a reading ORDER, not a decoration: a poster is read out of order unless it says
% otherwise. It rides ON the icon's corner (a disc centred on `bh.north east') rather than sitting
% beside it, so the pair reads as one mark -- and so a card's identity is still one object, not two.
% Numbering is 1..6 in reading order: objectives, approach, key results, predictions, interpretations,
% model comparison. Renumbering means editing all six call sites; there is no central list.
\newcommand{\blockheading}[6][\headingsize]{%
  \noindent
  \raisebox{-0.5\height}{%
    \tikz{%
      \node[fill=#2!16, rounded corners=9mm, minimum size=37mm, inner sep=0pt, text=#2,
        font=\iconxxl] (bh) {#3};
      \node[circle, fill=#2, text=white, minimum size=17mm, inner sep=0pt,
        font=\headingfont\bfseries\iconmd] at (bh.north east) {#4};}}%
  \hspace{9mm}%
  \begin{minipage}[c]{\dimexpr0.9\linewidth-50mm}
    {\kickerfont\cardtitlesize\color{#2}\MakeUppercase{#5}}\par
    \vspace{1.5mm}
    {\headingfont#1\color{textprimary}#6}\par
  \end{minipage}%
  \vspace{12mm}\par
}

% ===== Inline citation + supporting-detail reference tag ==========================================
% Literature citation embedded at point of use, small. It takes the CHROME turquoise -- the SAME
% colour as the \seebottom supporting-detail tags below -- because the two are the same kind of mark:
% a pointer AWAY from the poster (to the literature, to the supplement). They are deliberately not
% two colours. The former semantic coral (citecoral) is retired in colors.tex.
\newcommand{\litcite}[1]{{\color{accentturq}\notesize (#1)}}
% Small numbered badge (turquoise rounded square) matching the numbered points in supplement.tex.
% The badge is placed
% with \raisebox (a LaTeX box lift) around a picture whose baseline is pinned to the node CENTRE via
% a node anchor -- neither depends on the surrounding picture's axes, so the badge aligns identically
% in ordinary text and inside a flipped-axis timeline node (e.g. the Approach, x=1cm y=-1cm).
\newcommand{\refnum}[1]{\raisebox{3.2mm}{\tikz[baseline=(rn.center)]{\node(rn)[fill=accentturq,
  text=white, rounded corners=3mm, minimum size=12mm, inner sep=0pt,
  font=\headingfont\notesize]{#1};}}}
% Pointer(s) to numbered item(s) in the standalone supplement: a small QR-code glyph + the item
% number(s), signalling "scan the header QR to reach supplemental item N". The number badges are kept.
\newcommand{\seebottom}[1]{{\color{accentturq}\notesize \mbox{\icQR\;\refnum{#1}}}}
\newcommand{\seebottomtwo}[2]{{\color{accentturq}\notesize \mbox{\icQR\;\refnum{#1}\,\refnum{#2}}}}

% ===== TikZ arrow / connector styles ==============================================================
\tikzset{
  flowarrow/.style={-{Stealth[length=19pt,width=15pt]}, line width=2.2pt,
    rounded corners=8pt},
  softarrow/.style={-{Stealth[length=15pt,width=11pt]}, line width=1.6pt},
  thinarrow/.style={-{Stealth[length=11pt,width=8pt]}, line width=1.0pt},
  guideflow/.style={-{Stealth[length=22pt,width=16pt]}, line width=3pt,

  % A COORDINATE MARK: the point on a vector at which a coefficient is read off (alpha_k u_k, beta_k k_k,

  % the WHITE RING is what makes it a mark rather than a thicker piece of the arrow. The ring's line width
  % is absolute (TikZ does not scale line widths), so the mark keeps the same rim in every figure; the
  % RADIUS, however, IS scaled by the picture's `scale', so each figure passes a radius that lands at
  % ~4.4pt on the page (2.4pt at scale 1.85, 4pt at scale 1.1, ...). Use as: \draw[coordmark, fill=compred].
  coordmark/.style={draw=white, line width=1.2pt},
}
